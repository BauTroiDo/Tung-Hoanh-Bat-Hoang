cmake_minimum_required(VERSION 3.20)
project(HHTT_PortableCore LANGUAGES CXX)
include(CTest)
find_package(Threads REQUIRED)
add_library(hhtt_portable_core STATIC src/PortableCore.cpp src/WorldRuntime.cpp src/SpatialAoi.cpp src/AoiReplication.cpp)
target_link_libraries(hhtt_portable_core PUBLIC Threads::Threads)
target_include_directories(hhtt_portable_core PUBLIC include)
target_compile_features(hhtt_portable_core PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(hhtt_portable_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(hhtt_portable_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  add_executable(hhtt_portable_tests tests/portable_core_tests.cpp)
  target_link_libraries(hhtt_portable_tests PRIVATE hhtt_portable_core)
  add_test(NAME portable_core_tests COMMAND hhtt_portable_tests)
  add_executable(hhtt_portable_property_tests tests/portable_core_property_tests.cpp)
  target_link_libraries(hhtt_portable_property_tests PRIVATE hhtt_portable_core)
  add_test(NAME portable_core_property_tests COMMAND hhtt_portable_property_tests)
  add_executable(hhtt_world_runtime_tests tests/world_runtime_tests.cpp)
  target_link_libraries(hhtt_world_runtime_tests PRIVATE hhtt_portable_core)
  add_test(NAME world_runtime_tests COMMAND hhtt_world_runtime_tests)
  add_executable(hhtt_spatial_aoi_tests tests/spatial_aoi_tests.cpp)
  target_link_libraries(hhtt_spatial_aoi_tests PRIVATE hhtt_portable_core)
  add_test(NAME spatial_aoi_tests COMMAND hhtt_spatial_aoi_tests)
  add_executable(hhtt_aoi_replication_tests tests/aoi_replication_tests.cpp)
  target_link_libraries(hhtt_aoi_replication_tests PRIVATE hhtt_portable_core)
  add_test(NAME aoi_replication_tests COMMAND hhtt_aoi_replication_tests)
endif()
