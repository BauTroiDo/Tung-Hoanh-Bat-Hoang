cmake_minimum_required(VERSION 3.16)
project(jx3d_world_tick_bridge LANGUAGES CXX)
include(CTest)

add_library(jx3d_tick_bridge STATIC
  ../../gameserver/Sources/Core/Src/Jx3dLegacyWorldTickBridge.cpp)
target_compile_features(jx3d_tick_bridge PUBLIC cxx_std_11)
if(MSVC)
  target_compile_options(jx3d_tick_bridge PRIVATE /W4 /WX)
else()
  target_compile_options(jx3d_tick_bridge PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

if(BUILD_TESTING)
  add_executable(jx3d_tick_bridge_smoke world_tick_bridge_test.cpp)
  target_compile_features(jx3d_tick_bridge_smoke PRIVATE cxx_std_11)
  target_link_libraries(jx3d_tick_bridge_smoke PRIVATE jx3d_tick_bridge)
  if(MSVC)
    target_compile_options(jx3d_tick_bridge_smoke PRIVATE /W4 /WX)
  else()
    target_compile_options(jx3d_tick_bridge_smoke PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME legacy_world_tick_bridge_smoke COMMAND jx3d_tick_bridge_smoke)
  add_test(NAME legacy_world_snapshot_overquota COMMAND jx3d_tick_bridge_smoke overlimit)
  add_test(NAME legacy_world_snapshot_null_data COMMAND jx3d_tick_bridge_smoke null-data)
endif()
