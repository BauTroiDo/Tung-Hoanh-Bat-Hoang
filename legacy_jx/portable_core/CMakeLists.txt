cmake_minimum_required(VERSION 3.20)
project(HHTT_PortableCore LANGUAGES CXX)
include(CTest)
add_library(hhtt_portable_core STATIC src/PortableCore.cpp)
target_include_directories(hhtt_portable_core PUBLIC include)
target_compile_features(hhtt_portable_core PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(hhtt_portable_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(hhtt_portable_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  add_executable(hhtt_portable_tests tests/portable_core_tests.cpp)
  target_link_libraries(hhtt_portable_tests PRIVATE hhtt_portable_core)
  add_test(NAME portable_core_tests COMMAND hhtt_portable_tests)
endif()
